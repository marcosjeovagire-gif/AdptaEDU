export interface MetricasAlunoDashboard {
  estudante: Estudante;
  totalAtividades: number;
  atividadesConcluidas: number;

  alinhadasAoPei: number;

export function calcularMetricasEstudantes(
  estudantes: Estudante[],
  atividades: AtividadeAdaptada[],
  registrosApoio: RegistroAcompanhamentoApoio[] = [],
  turmaFiltro: string = 'todas',
  dataInicio?: string,
  dataFim?: string
): {
  alunosMetricas: MetricasAlunoDashboard[];
  resumoGeral: {
    totalAlunos: number;
    totalAtividades: number;
    totalConcluidas: number;
    taxaMediaConclusao: number;
    taxaAlinhamentoPei: number;
    turmasPresentes: string[];
  };
} {
  const atividadesFiltradas = atividades.filter(ativ => {
    if (turmaFiltro !== 'todas' && ativ.turma !== turmaFiltro) return false;
    if (dataInicio || dataFim) {
      const ativData = ativ.dataCriacao || '';
      let dataIso = '';
      if (ativData.includes('/')) {
        const [d, m, y] = ativData.split('/');
        dataIso = `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`;
      } else {
        dataIso = ativData;
      }
      if (dataInicio && dataIso < dataInicio) return false;
      if (dataFim && dataIso > dataFim) return false;
    }
    return true;
  });

  const estudantesFiltrados = estudantes.filter(e => {
    if (turmaFiltro === 'todas') return true;
    return e.turma === turmaFiltro || e.anoSerie === turmaFiltro;
  });

  const alunosMetricas: MetricasAlunoDashboard[] = estudantesFiltrados.map(est => {
    const ativsDoAluno = atividadesFiltradas.filter(a => a.estudanteId === est.id);
    const concluidas = ativsDoAluno.filter(a => a.status === 'Revisada' || (a as any).concluida === true);
    const alinhadasPei = ativsDoAluno.filter(a => a.alinhadaAoPei);
    const taxa = ativsDoAluno.length > 0 ? Math.round((concluidas.length / ativsDoAluno.length) * 100) : 0;

    const logsAluno = registrosApoio.filter(r => r.studentId === est.id);
    let nivelEngajamento = 'Participativo';
    if (logsAluno.length > 0) {
      const comDificuldade = logsAluno.filter(l => 
        l.conseguiuRealizarAtividade === 'Não' || 
        l.conseguiuRealizarAtividade === 'Parcialmente' || 
        l.demonstrouIrritabilidadeOuSinais === 'Sim' || 
        l.nivelAutonomia === 'Com Mediação Física/Visual' || 
        l.nivelAutonomia === 'Não Realizou'
      ).length;
      if (comDificuldade > logsAluno.length / 2) nivelEngajamento = 'Apoio Constante';
      else if (comDificuldade > 0) nivelEngajamento = 'Mediação Pontual';
      else nivelEngajamento = 'Autônomo c/ Apoio';
    } else if (taxa >= 80) {
      nivelEngajamento = 'Alto Engajamento';
    } else if (taxa >= 40) {
      nivelEngajamento = 'Médio Engajamento';
    } else if (ativsDoAluno.length > 0) {
      nivelEngajamento = 'Necessita Estímulo';
    } else {
      nivelEngajamento = 'Sem Atividades';
    }

    const laudoLow = `${est.laudo || ''} ${est.observacoesAcessibilidade || ''}`.toLowerCase();
    const formatoLetra: 'bastao' | 'cursiva' = 
      (est as any).formatoLetra === 'cursiva' ? 'cursiva' :
      (laudoLow.includes('tea') || laudoLow.includes('intelect') || laudoLow.includes('motora') || laudoLow.includes('disgraf') ? 'bastao' : 'bastao');

    const ultimaData = ativsDoAluno.length > 0 ? ativsDoAluno[ativsDoAluno.length - 1].dataCriacao : undefined;

    return {
      estudante: est,
      totalAtividades: ativsDoAluno.length,
      atividadesConcluidas: concluidas.length,
      taxaConclusao: taxa,
      alinhadasAoPei: alinhadasPei.length,
      nivelEngajamento,
      formatoLetra,
      ultimaAtividadeData: ultimaData,
    };
  });

  const totalAtividades = atividadesFiltradas.length;
  const totalConcluidas = atividadesFiltradas.filter(a => a.status === 'Revisada' || (a as any).concluida === true).length;
  const totalAlinhadas = atividadesFiltradas.filter(a => a.alinhadaAoPei).length;
  const taxaMediaConclusao = totalAtividades > 0 ? Math.round((totalConcluidas / totalAtividades) * 100) : 0;
  const taxaAlinhamentoPei = totalAtividades > 0 ? Math.round((totalAlinhadas / totalAtividades) * 100) : 0;
  const turmasPresentes = Array.from(new Set(estudantesFiltrados.map(e => e.turma || e.anoSerie || 'Não informada'))).sort();

  return {
    alunosMetricas,
    resumoGeral: {
      totalAlunos: estudantesFiltrados.length,
      totalAtividades,
      totalConcluidas,
      taxaMediaConclusao,
      taxaAlinhamentoPei,
      turmasPresentes,
    },
  };
}

## ETAPA 1 — CORREÇÃO DE BUGS

1.1. **`sanitizarNomeArquivo` importado mas não usado**: aplique-o em `turmaSelecionada` antes de compor `fileName`. Se ele já cobre caracteres inválidos, adicione pós-processamento extra (strip `/ \ : * ? " < > |`, limite 120 chars, timestamp ISO compacto `YYYYMMDD-HHMMSS`).

1.2. **Lógica morta em `formatoLetra`**: o ternário aninhado sempre retorna `'bastao'` no segundo ramo. Simplifique para a regra determinística:
```ts
const formatoLetra: 'bastao' | 'cursiva' =
  (est as any).formatoLetra === 'cursiva' ? 'cursiva' : 'bastao';
```
e extraia para função pura `inferirFormatoLetra(estudante): 'bastao' | 'cursiva'` exportada.

1.3. **Parsing de data frágil** em `calcularMetricasEstudantes`:
- Valide `d`, `m`, `y` antes de `padStart`. Se inválida, retorne `false` no filtro (registre contagem de descartadas para auditoria se quiser).
- Aceite também ISO com hora. Torne `dataFim` inclusiva (comparar `<=` ou concatenar `T23:59:59.999Z`).
- Extraia helper puro `normalizarDataParaIso(input?: string): string | null`.

1.4. **`perfisPedagogicos` nunca usado**: use-o para enriquecer o parecer pedagógico (ex.: linha extra "Perfis pedagógicos mapeados: X alunos nível N1, Y nível N2..."). Se não houver dados, degrade graciosamente sem quebrar layout. Alternativamente documente com `// reservado para uso futuro` e remova do uso — mas prefira USAR.

1.5. **`checkPageBreak` não protege conteúdo maior que uma página**: se `neededHeight > pageHeight - 16 - 12`, o conteúdo estoura. Adicione salvaguarda: função `ensureSpace(needed, drawFn)` que (a) chama `checkPageBreak`, (b) se ainda não cabe, divide por linhas ou reduz fonte progressivamente até caber, mantendo legibilidade mínima (≥ 5pt).

1.6. **Consistência `currentY` vs `renderMiniCabecalho` vs `checkPageBreak`**: garanta ordem determinística — `checkPageBreak` reseta `currentY = 12` **antes** de chamar `renderMiniCabecalho`, e o mini cabeçalho soma `+7` uma única vez. Documente em comentário curto.

1.7. **`doc.save()` / `doc.addImage()` sem try/catch**: envolva, retorne `{ success: false, fileName: '', error: string }`. Estenda o tipo de retorno para `Promise<{ success: boolean; fileName: string; error?: string }>` (retrocompatível).

1.8. **`catch` vazio em `addImage`**: adicione `console.warn` condicional (`import.meta.env.DEV` no Vite ou `process.env.NODE_ENV !== 'production'`) e fallback silencioso em prod.

## ETAPA 2 — PRONTO PARA PRODUÇÃO EM REDE

2.1. **Não bloquear a thread principal** em turmas grandes (500+ alunos): renderize linhas em blocos de ~50 com `await new Promise(r => setTimeout(r, 0))` entre blocos. Exponha `onProgress?: (pct: number) => void` opcional em `ParametrosExportacaoDashboard` (retrocompatível) e chame a cada bloco.

2.2. **Nome de arquivo seguro para download em rede**: sanitize + limite de tamanho + timestamp ISO compacto. Garanta ASCII onde possível, preservando acentos com fallback (ex.: `normalize('NFD').replace(/[\u0300-\u036f]/g, '')` opcional).

2.3. **Memória**: use `doc.splitTextToSize` para nomes/laudos longos em vez de `slice` cego. Evite concatenar strings gigantes em loops — reutilize buffers curtos.
